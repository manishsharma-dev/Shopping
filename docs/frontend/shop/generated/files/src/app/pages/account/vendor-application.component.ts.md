import { Component, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { MatButtonModule } from '@angular/material/button';
import { MatFormFieldModule } from '@angular/material/form-field';
import { MatInputModule } from '@angular/material/input';
@Component({
  selector: 'app-vendor-application',
  standalone: true,
  imports: [FormsModule, MatButtonModule, MatFormFieldModule, MatInputModule],
  template: `
    <h2>Become a vendor</h2>
    <p>Submit your business for review by your state or district administrator.</p>
    @if (error()) {
      <p role="alert">{{ error() }}</p>
    }
    @for (a of applications(); track a.id) {
      <p>
        <strong>{{ a.name }}: {{ a.status }}</strong> {{ a.data.reason || '' }}
      </p>
    }
    @if (!applications().length || applications()[0].status === 'rejected') {
      <form #form="ngForm" (ngSubmit)="form.valid && submit()" style="display:grid;gap:12px">
        @for (field of fields; track field.key) {
          <mat-form-field appearance="outline"
            ><mat-label>{{ field.label }}</mat-label
            ><input
              matInput
              [name]="field.key"
              [(ngModel)]="draft[field.key]"
              required
              [maxlength]="field.max"
          /></mat-form-field>
        }
        <button mat-flat-button [disabled]="form.invalid || busy()">
          {{ busy() ? 'Please wait' : 'Submit application' }}
        </button>
      </form>
    }
  `,
})
export class VendorApplication {
  readonly busy = signal(false);
  readonly error = signal('');
  readonly applications = signal<any[]>([]);
  draft: Record<string, string> = {};
  fields = [
    { key: 'name', label: 'Business name', max: 150 },
    { key: 'state', label: 'State (official name)', max: 100 },
    { key: 'district', label: 'District (official name)', max: 100 },
    { key: 'phone', label: 'Business phone', max: 30 },
    { key: 'address', label: 'Business address', max: 1000 },
  ];
  constructor() {
    void this.load();
  }
  async request(data?: object) {
    const r = await fetch('http://localhost:3000/api/manage/' + (data ? 'apply' : 'applications'), {
      method: data ? 'POST' : 'GET',
      credentials: 'include',
      headers: data ? { 'Content-Type': 'application/json', 'X-Shopping-Client': 'web' } : {},
      body: data ? JSON.stringify({ data }) : undefined,
    });
    const result = await r.json();
    if (!r.ok) throw new Error(result.message || 'Request failed');
    return result;
  }
  async load() {
    try {
      this.applications.set(await this.request());
    } catch (e) {
      this.error.set((e as Error).message);
    }
  }
  async submit() {
    if (this.busy()) return;
    this.busy.set(true);
    this.error.set('');
    try {
      await this.request(this.draft);
      await this.load();
    } catch (e) {
      this.error.set((e as Error).message);
    } finally {
      this.busy.set(false);
    }
  }
}
